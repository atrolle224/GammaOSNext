# Pegasus game list for the RG477M: copy this ROMs folder to the root of the SD card,
# then add it in Pegasus (Start > Settings > Set game directories).
# Each system reads its own folder below; a system with no games stays hidden.
# A launch line can be swapped for the commented alternative under it.

collection: Game Boy
shortname: gb
directories: gb
extensions: gb, zip, 7z
launch: am start
  -n com.retroarch.rg477m/com.retroarch.browser.retroactivity.RetroActivityFuture
  -e ROM {file.path}
  -e LIBRETRO /data/data/com.retroarch.rg477m/cores/gambatte_libretro_android.so
  -e CONFIGFILE /storage/emulated/0/RetroArch/retroarch.cfg
  --activity-clear-task --activity-clear-top --activity-no-history
