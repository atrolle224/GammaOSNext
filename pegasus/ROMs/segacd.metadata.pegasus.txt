# Sega CD for Pegasus on the RG477M. Games go in ROMs/segacd.
# To use another app, swap the launch line for the commented one below it.
# game: entries for this system's games (title, description...) can go at the end.

collection: Sega CD
shortname: segacd
# every file outside ROMs/segacd is left to the other systems
ignore-regex: ^(?!.*/segacd/)
extensions: chd, cue, iso, m3u
launch: am start
  -n com.retroarch.rg477m/com.retroarch.browser.retroactivity.RetroActivityFuture
  -e ROM {file.path}
  -e LIBRETRO /data/data/com.retroarch.rg477m/cores/genesis_plus_gx_libretro_android.so
  -e CONFIGFILE /storage/emulated/0/RetroArch/retroarch.cfg
  --activity-clear-task --activity-clear-top --activity-no-history
