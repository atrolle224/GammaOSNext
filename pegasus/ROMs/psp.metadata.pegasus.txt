# PlayStation Portable for Pegasus on the RG477M. Games go in ROMs/psp.
# To use another app, swap the launch line for the commented one below it.
# game: entries for this system's games (title, description...) can go at the end.

collection: PlayStation Portable
shortname: psp
directories: psp
extensions: iso, cso, chd, pbp, elf
launch: am start
  -n org.ppsspp.ppsspp.rg477m/org.ppsspp.ppsspp.PpssppActivity
  -a android.intent.action.VIEW -c android.intent.category.DEFAULT
  -d {file.uri} -t application/octet-stream
  --activity-clear-task --activity-clear-top --activity-no-history
